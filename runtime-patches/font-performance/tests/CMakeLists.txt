cmake_minimum_required(VERSION 3.26)
project(FontDecoderEquivalence LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT DEFINED RUNTIME_SOURCE)
  message(FATAL_ERROR "RUNTIME_SOURCE must name the isolated beta.1.1 source directory")
endif()
set(DECODER ${RUNTIME_SOURCE}/runtime/three_ds_recomp/src/fast/oot3d/pica_texture_decode.cpp)
add_library(reference OBJECT ${RUNTIME_SOURCE}/runtime/three_ds_recomp/src/fast/oot3d/pica_texture_decode.reference.cpp)
target_compile_definitions(reference PRIVATE Oot3d=ReferenceOot3d)
add_library(candidate OBJECT ${DECODER})
foreach(target reference candidate)
  target_include_directories(${target} PRIVATE ${RUNTIME_SOURCE}/runtime/three_ds_recomp/include)
endforeach()
add_executable(decoder_equivalence decoder_equivalence.cpp $<TARGET_OBJECTS:reference> $<TARGET_OBJECTS:candidate>)
target_include_directories(decoder_equivalence PRIVATE ${RUNTIME_SOURCE}/runtime/three_ds_recomp/include)
